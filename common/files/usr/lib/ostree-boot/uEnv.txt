ostree_load_kernel=load ${rootdev}:${rootpart} ${kernel_addr_r} /root${kernel_image}
ostree_save_kernel_size=setenv kernel_size ${filesize}
ostree_load_ramdisk=load ${rootdev}:${rootpart} ${ramdisk_addr_r} /root${ramdisk_image}
ostree_save_ramdisk_size=setenv ramdisk_size ${filesize}
ostree_load_fdt=load ${rootdev}:${rootpart} ${fdt_addr_r} /root${fdtdir}/${fdtfile}
ostree_boot=bootefi ${kernel_addr_r}:${kernel_size} ${ramdisk_addr_r}:${ramdisk_size} ${fdt_addr_r}
uenvcmd=run ostree_load_kernel ostree_save_kernel_size ostree_load_ramdisk ostree_save_ramdisk_size ostree_load_fdt ostree_boot
ostree_select_kernel=env indirect kernel_image kernel_image${ostree_index}
ostree_select_ramdisk=env indirect ramdisk_image ramdisk_image${ostree_index}
ostree_select_fdt=env indirect fdtdir fdtdir${ostree_index}
ostree_select_bootargs=env indirect bootargs bootargs${ostree_index}
ostree_select_deployment=run ostree_select_kernel ostree_select_ramdisk ostree_select_fdt ostree_select_bootargs
uenvcmd_deployment=run ostree_select_deployment uenvcmd
